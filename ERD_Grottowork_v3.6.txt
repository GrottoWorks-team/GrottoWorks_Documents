// Database Design: PostgreSQL-oriented DBML
// v3.6 (14/10/2026): season created in PLANNING (no DRAFT); unique season name
// and year per parish; status notes aligned with API Contract v1.3.
// v3.5 (07/10/2026): team decisions D-01..D-06 -- leader approves purchase
// requests and assigns buyers through PURCHASE tasks; donation pledge workflow
// restored with per-receipt posting; support requests cover staffing and/or
// material (stock transfer between work areas); no readiness template; volunteer

// notification, notification_preference, activity_log, file_object.
// See v3.4 -> v3.5 change notes at the end of this file.
// v3.3: staff-entered donor profiles; MONEY/MATERIAL donations;
// donation is the authoritative money source; contribution links to donation.
// Previous v3.2: single role; monetary amount; multi-position/multi-skill support;
// manual task completion approval; multi-material purchase requests.
// DB constraints cover local checks and references. Cross-table business
// rules in Notes must be enforced by the application in transactions.
// Draft requests may be empty; submitted support/purchase requests must
// contain at least one position/item. Validate in the service.

  indexes {

Table task {
  task_id uuid [pk]

  work_area_id uuid [not null]
  created_by_user_id uuid [not null]

  task_type varchar(30) [not null, default: 'WORK', note: 'WORK, PURCHASE, CORRECTIVE']
  purchase_request_id uuid [note: 'Required when task_type = PURCHASE: the approved purchase request this buying task fulfils (D-01).']

  task_name varchar(200) [not null]
  task_description text
  completion_criteria text [note: 'Leader-defined completion criteria shown in the review package.']
  task_priority varchar(30) [not null]
  estimated_hours decimal(6,2)
  required_volunteer_count int
  start_date date
  deadline date
  task_status varchar(30) [not null, note: 'DRAFT, OPEN, NOT_STARTED, IN_PROGRESS, SUBMITTED, REVISION_REQUIRED, COMPLETED, CANCELLED']

  created_at timestamp [not null]
  updated_at timestamp

  checks {
    `(task_type = 'PURCHASE' AND purchase_request_id IS NOT NULL) OR (task_type <> 'PURCHASE' AND purchase_request_id IS NULL)` [name: 'task_purchase_link']
  }
  Note: 'PURCHASE tasks are created by the work-area leader only after the purchase request is APPROVED. The buyer is the volunteer whose assignment on this task is ACCEPTED. A PURCHASE task becomes COMPLETED automatically when the material officer confirms delivery of its purchase (BR-65). CORRECTIVE tasks are linked from readiness_item.corrective_task_id.'
}

Table support_request {
  support_request_id uuid [pk]
  season_id uuid [not null]
  work_area_id uuid [not null]
  requested_by_user_id uuid [not null]
  support_type varchar(30) [not null, note: 'STAFFING, MATERIAL, MIXED']

  Note: 'D-03: a request may ask for people (positions) and/or materials. STAFFING needs >= 1 position, MATERIAL needs >= 1 material line, MIXED needs both when published (BR-62). Total headcount is the sum of position counts, not skill counts. Validate that season matches work area.'

  support_request_position_id uuid [pk]

Table support_response {
  support_response_id uuid [pk]

  support_request_id uuid [not null]
  volunteer_user_id uuid
  responding_community_id uuid

  response_status varchar(30) [not null, note: 'PENDING, PARTIALLY_ACCEPTED, ACCEPTED, REJECTED, CANCELLED']
  response_quantity int [note: 'Offered headcount only; actual named participants are in support_response_member. Do not add these two counts. Material offers are in support_response_material.']
  response_note text

  created_at timestamp [not null]
}


Table support_response_member {
  support_response_member_id uuid [pk]
  support_response_id uuid [not null]
  support_request_position_id uuid [not null]
  volunteer_user_id uuid [not null]
  member_status varchar(30) [not null, note: 'PENDING, APPROVED, REJECTED, CANCELLED, COMPLETED']

  indexes {
    (support_response_id, volunteer_user_id) [unique]
  }
  Note: 'Service must validate response and position belong to the same request, all required skills, capacity, and at most one active position per person in that request, including across responses. Individual response volunteer must match its member; community response can name multiple members.'

Table support_request_material {
  support_request_material_id uuid [pk]
  support_request_id uuid [not null]
  material_id uuid [not null]

  required_quantity decimal(12,2) [not null]
  material_note text

  indexes {
    (support_request_id, material_id) [unique]
  }
  checks {
    `required_quantity > 0` [name: 'support_request_material_positive_quantity']
  }
}

Table support_response_material {
  support_response_material_id uuid [pk]
  support_response_id uuid [not null]
  support_request_material_id uuid [not null]
  source_work_area_id uuid [not null, note: 'Work area whose stock supplies the material (supporting community).']

  offered_quantity decimal(12,2) [not null]
  approved_quantity decimal(12,2)
  received_usable_quantity decimal(12,2)
  received_unusable_quantity decimal(12,2)

  material_status varchar(30) [not null, note: 'PENDING, APPROVED, REJECTED, TRANSFERRED, CANCELLED']
  approved_by_user_id uuid
  approved_at timestamp
  received_confirmed_by_user_id uuid
  received_confirmed_at timestamp
  material_note text

  checks {
    `offered_quantity > 0` [name: 'support_response_material_positive_offer']
    `approved_quantity IS NULL OR (approved_quantity > 0 AND approved_quantity <= offered_quantity)` [name: 'support_response_material_valid_approval']
    `received_usable_quantity IS NULL OR received_usable_quantity >= 0` [name: 'support_response_material_nonnegative_usable']
    `received_unusable_quantity IS NULL OR received_unusable_quantity >= 0` [name: 'support_response_material_nonnegative_unusable']
  }
  Note: 'Material support = stock transfer between work areas (D-03). Offer must not exceed unallocated stock of source_work_area_id at approval time (BR-63). On approval the service posts TRANSFER_OUT on the source work area; the receiving material officer confirms received quantities, then TRANSFER_IN posts only usable quantity (BR-64). Source and target work areas must belong to the same season and must be different. Individual volunteers donating materials use the donation pledge workflow instead.'
}

Table purchase_request {
  purchase_request_id uuid [pk]
  work_area_id uuid [not null]
  requested_by_user_id uuid [not null]
  approved_by_user_id uuid [note: 'Work-area leader who approved or rejected (D-01).']
  request_reason text
  request_status varchar(30) [not null, note: 'DRAFT, PENDING_APPROVAL, APPROVED, REJECTED, BUYER_ASSIGNED, PURCHASED, PARTIALLY_RECEIVED, COMPLETED, CANCELLED']
  decision_note text [note: 'Required when REJECTED (BR-31).']

  rejected_at timestamp

  Note: 'Created by the material officer, approved by the leader of the same work area (BR-28, BR-29).   purchase_id uuid [pk]

  purchase_request_id uuid
  task_id uuid [note: 'PURCHASE task that produced this purchase. Required when purchase_request_id is set.']
  buyer_user_id uuid [not null, note: 'Volunteer with the ACCEPTED assignment on task_id.']
  confirmed_by_user_id uuid [note: 'Material officer who confirmed delivery and usable quantity.']

  supplier_name varchar(150)
  purchase_date date [not null]
  total_cost decimal(14,2)

  purchase_status varchar(30) [not null, note: 'SUBMITTED, NEEDS_INFO, CONFIRMED, CANCELLED']

  confirmed_at timestamp
  cost_overrun_approved_by_user_id uuid [note: 'Leader who approved actual cost above the approved estimate (BR-32). Required before confirmation when total_cost exceeds the request estimate.']
  cost_overrun_approved_at timestamp

  Note: 'Receipt images/PDFs are file_object rows with reference_type = PURCHASE_RECEIPT and reference_id = purchase_id (max 5). SUBMITTED does not increase stock (BR-26); delivery confirmation posts PURCHASE_DELIVERY transactions for usable quantity only.'
}

Table purchase_item {
  purchase_item_id uuid [pk]
  purchase_request_item_id uuid [note: 'Required when purchase has a request. Service validates same request as purchase header and same material. NULL for purchases without a request.']

  purchase_id uuid [not null]
  material_id uuid [not null]

  purchased_quantity decimal(12,2) [not null]
  unit_price decimal(14,2) [not null]

  received_usable_quantity decimal(12,2) [note: 'Set by the material officer on delivery confirmation.']
  received_unusable_quantity decimal(12,2)

  checks {
    `purchased_quantity > 0 AND unit_price >= 0 AND total_price >= 0` [name: 'purchase_item_valid_amounts']
    `received_usable_quantity IS NULL OR received_usable_quantity >= 0` [name: 'purchase_item_nonnegative_usable']
  }
}


// ============================================================
// 5. DONOR & DONATION (PLEDGE + RECEIPTS)
// ============================================================

Enum donation_kind {
  MONEY
  MATERIAL
}

Table donor {
  donor_id uuid [pk]
  user_id uuid [unique, note: 'Optional verified account link. External donors have no account and no role.']
  donor_name varchar(150) [not null]
  phone varchar(30)
  email varchar(255)
  address varchar(255)
  donor_note text
  created_by_user_id uuid [not null]
  created_at timestamp [not null]
  updated_at timestamp

  Note: 'Officer/Admin enters or selects donor on the internal form. A volunteer pledging through the app gets (or reuses) a donor row linked by user_id. Do not auto-merge by name or unverified phone/email. Staff account identifies the recorder, never the donor. Contact details are optional for direct receipt.'
}

Enum donation_status {
  PLEDGED
  PARTIALLY_RECEIVED
  RECEIVED
  CANCELLED
}

Table donation {
  donation_id uuid [pk]
  donor_id uuid [not null]
  season_id uuid [not null]
  work_area_id uuid [note: 'Receiving work area. Required for MATERIAL (stock is per work area, BR-24); optional for MONEY.']
  donation_type donation_kind [not null]
  donation_status donation_status [not null]
  pledged_amount decimal(14,2) [note: 'MONEY pledge in VND; NULL for direct receipts and for MATERIAL.']
  amount decimal(14,2) [note: 'MONEY: total actually received (sum of donation_receipt.received_amount). NULL for MATERIAL.']
  donation_date timestamp [not null, note: 'Pledge date, or receipt date for direct records.']
  created_by_user_id uuid [not null, note: 'Volunteer who pledged or officer/admin who recorded.']
  created_at timestamp [not null]
  updated_at timestamp
  cancel_reason text
  donation_note text

  checks {
    `(donation_type = 'MONEY' AND (pledged_amount IS NULL OR pledged_amount > 0) AND (amount IS NULL OR amount >= 0)) OR (donation_type = 'MATERIAL' AND amount IS NULL AND pledged_amount IS NULL)` [name: 'donation_type_amount_consistency']
    `donation_type <> 'MATERIAL' OR work_area_id IS NOT NULL` [name: 'donation_material_requires_work_area']
    `donation_status <> 'CANCELLED' OR cancel_reason IS NOT NULL` [name: 'donation_cancel_reason']
  }
  Note: 'D-02: a donation starts either as a volunteer PLEDGE (status PLEDGED, nothing posted) or as an officer/admin direct record (status RECEIVED with one receipt created in the same transaction). Each physical receipt is a donation_receipt; status moves PLEDGED -> PARTIALLY_RECEIVED -> RECEIVED, or to CANCELLED with a reason (cancelling keeps already received receipts). Pledged values never count as stock or money received (BR-39). Legacy v3.4 rule kept below for direct records: Service enforces staff role and parish scope. MONEY has no donation_item. MATERIAL must have at least one item with positive received_quantity and at least one unit received across the donation; create header and items atomically. Each row represents one actual receipt occasion; later deliveries create new donations for the same donor. Money is authoritative here, never duplicated in contribution. Material usable stock is posted exactly once with this record. Changes to recorded quantities require atomic reconciliation with stock and contribution summaries.'
}


Table donation_item {
  donation_item_id uuid [pk]
  donation_id uuid [not null]
  material_id uuid [not null]
  pledged_quantity decimal(12,2) [note: 'Promised quantity for a pledge; NULL for direct unpledged receipt.']
  received_quantity decimal(12,2) [not null, default: 0, note: 'Cached total of donation_receipt_item.received_quantity, including unusable units.']
  unusable_quantity decimal(12,2) [not null, default: 0, note: 'Cached total of donation_receipt_item.unusable_quantity.']
  item_note text

  indexes {
    (donation_id, material_id) [unique]
  }
  checks {
    `pledged_quantity IS NULL OR pledged_quantity > 0` [name: 'donation_item_positive_pledge']
    `received_quantity >= 0 AND unusable_quantity >= 0 AND unusable_quantity <= received_quantity` [name: 'donation_item_valid_receipt']
  }
  Note: 'Only for MATERIAL donation; enforce parent type in service or trigger. Quantities use material.material_unit, no independent donation unit. Usable received quantity = received_quantity - unusable_quantity. Stock is posted per donation_receipt_item, never from this row directly. pledged_quantity is the promised amount of a pledge; leave NULL for direct receipts. Item receipt status (PLEDGED, PARTIALLY_RECEIVED, FULLY_RECEIVED, UNUSABLE) is derived from pledged/received/unusable quantities instead of independently stored.'
}

Table donation_receipt {
  donation_receipt_id uuid [pk]
  donation_id uuid [not null]

  received_at timestamp [not null]
  received_by_user_id uuid [not null, note: 'Material officer or admin who physically received it.']
  received_amount decimal(14,2) [note: 'MONEY only; NULL for MATERIAL.']
  receipt_note text
  created_at timestamp [not null]

  checks {
    `received_amount IS NULL OR received_amount > 0` [name: 'donation_receipt_positive_amount']
  }
  Note: 'One physical receipt occasion for a donation. Header, items, stock posting (DONATION_RECEIPT_ITEM transactions for usable quantity only), donation cached totals/status and contribution upsert run in one transaction (BR-38). Evidence photos are file_object rows with reference_type = DONATION_RECEIPT.'
}

Table donation_receipt_item {
  donation_receipt_item_id uuid [pk]
  donation_receipt_id uuid [not null]
  donation_item_id uuid [not null]

  received_quantity decimal(12,2) [not null]
  unusable_quantity decimal(12,2) [not null, default: 0]
  item_note text

  indexes {
    (donation_receipt_id, donation_item_id) [unique]
  }
  checks {
    `received_quantity > 0 AND unusable_quantity >= 0 AND unusable_quantity <= received_quantity` [name: 'donation_receipt_item_valid_quantity']
  }
  Note: 'donation_item must belong to the same donation as the receipt. Usable = received_quantity - unusable_quantity. Receiving more than pledged is allowed with a warning and recorded as actually received.'
}

Table borrowed_item {
  borrowed_item_id uuid [pk]

  offered_by_user_id uuid [not null, note: 'Volunteer (or officer) who entered the loan offer, possibly on behalf of the lender (D-05). Contact point inside the system.']
  accepted_by_user_id uuid [note: 'Material officer who accepted the offer.']
  work_area_id uuid
  task_id uuid [note: 'Optional task the item is assigned to.']
  season_id uuid [not null]

  owner_name varchar(150) [not null, note: 'Lender name; the lender does not need an account.']
  owner_phone varchar(30)
  owner_address varchar(255)
  owner_note text

  borrowed_item_name varchar(150) [not null]
  borrowed_item_quantity decimal(12,2) [not null]
  borrowed_item_description text
  initial_condition varchar(50)
  expected_return_date date [not null]
  borrowed_item_status varchar(30) [not null, note: 'OFFERED, ACCEPTED, REJECTED, ISSUED, IN_USE, RETURNED. OVERDUE is derived from expected_return_date.']
  accepted_at timestamp
  created_at timestamp [not null]

  checks {
