// ============================================================
// GROT TOWORKS - ERD v3.4
// GrottoWorks - A Parish Christmas Grotto and Decoration
// Management System
//
// Database Design: PostgreSQL-oriented DBML
// v3.3: staff-entered donor profiles; MONEY/MATERIAL donations;
// donation is the authoritative money source; contribution links to donation.
// Previous v3.2: single role; monetary amount; multi-position/multi-skill support;
// manual task completion approval; multi-material purchase requests.
// DB constraints cover local checks and references. Cross-table business
// rules in Notes must be enforced by the application in transactions.
// Draft requests may be empty; submitted support/purchase requests must
// contain at least one position/item. Validate in the service.
// Corrected one-to-one DBML FK direction: volunteer_profile references app_user.
// ============================================================


// ============================================================
// 1. ORGANIZATION & USER MANAGEMENT
// ============================================================

Table parish {
  parish_id uuid [pk]
  parish_name varchar(150) [not null]
  parish_address varchar(255)
  parish_description text
  parish_status varchar(30) [not null]

  created_at timestamp [not null]
  updated_at timestamp
}

  community_id uuid [pk]
  parish_id uuid [not null]

  community_name varchar(150) [not null]
  community_description text
  community_status varchar(30) [not null]

  created_at timestamp [not null]
  updated_at timestamp
}

Table app_user {
  user_id uuid [pk]

  full_name varchar(150) [not null]

  role_id uuid [pk]

  role_code varchar(50) [not null, unique]

  support_response_member_id uuid [pk]

    (support_response_id, volunteer_user_id) [unique]
  }
  Note: 'Service must validate response and position belong to the same request, all required skills, capacity, and at most one active position per person in that request, including across responses. Individual response volunteer must match its member; community response can name multiple members.'

Table purchase_request {
  purchase_request_id uuid [pk]
  work_area_id uuid [not null]
  requested_by_user_id uuid [not null]
  approved_by_user_id uuid
  request_reason text
  request_status varchar(30) [not null]
  approved_at timestamp
  created_at timestamp [not null]

  Note: 'Header for multiple requested materials. Approval applies to the whole request. Estimated total is SUM of item estimated_cost; do not store a second editable total.'
}

Table purchase_request_item {
  purchase_request_item_id uuid [pk]
  purchase_request_id uuid [not null]
  material_id uuid [not null]
  requested_quantity decimal(12,2) [not null]
  estimated_cost decimal(14,2) [note: 'Estimated total cost of this line, not unit price. NULL means not estimated.']
  item_note text

  indexes {
    (purchase_request_id, material_id) [unique]
  }
  checks {
    `requested_quantity > 0` [name: 'purchase_request_item_positive_quantity']
    `estimated_cost IS NULL OR estimated_cost >= 0` [name: 'purchase_request_item_nonnegative_cost']
  }
}

Table purchase {
  purchase_id uuid [pk]

  purchase_request_id uuid
  buyer_user_id uuid [not null]

  supplier_name varchar(150)
  purchase_date date [not null]

}

Table purchase_item {
  purchase_item_id uuid [pk]

  purchase_id uuid [not null]

  purchased_quantity decimal(12,2) [not null]
  unit_price decimal(14,2) [not null]

// ============================================================
// 5. DONOR & DONATION (STAFF-ENTERED)
// ============================================================

Enum donation_kind {
  MONEY
  MATERIAL
}

Table donor {
  donor_id uuid [pk]
  user_id uuid [unique, note: 'Optional verified account link. External donors have no account and no role.']
  donor_name varchar(150) [not null]
  phone varchar(30)
  email varchar(255)
  address varchar(255)
  donor_note text
  created_by_user_id uuid [not null]
  created_at timestamp [not null]
  updated_at timestamp

  Note: 'Officer/Admin enters or selects donor on the internal form. Do not auto-merge by name or unverified phone/email. Staff account identifies the recorder, never the donor. Contact details are optional for direct receipt.'
}

Table donation {
  donation_id uuid [pk]
  donor_id uuid [not null]
  season_id uuid [not null]
  donation_type donation_kind [not null]
  amount decimal(14,2) [note: 'Actual money received in VND for MONEY; NULL for MATERIAL.']
  donation_date timestamp [not null]
  created_by_user_id uuid [not null]
  created_at timestamp [not null]
  donation_note text

  checks {
    `(donation_type = 'MONEY' AND amount IS NOT NULL AND amount > 0) OR (donation_type = 'MATERIAL' AND amount IS NULL)` [name: 'donation_type_amount_consistency']
  }
  Note: 'Officer/Admin records actual received donations through an internal form. No pledge, draft, approval, receipt-document, or status workflow on this header. Service enforces staff role and parish scope. MONEY has no donation_item. MATERIAL must have at least one item with positive received_quantity and at least one unit received across the donation; create header and items atomically. Each row represents one actual receipt occasion; later deliveries create new donations for the same donor. Money is authoritative here, never duplicated in contribution. Material usable stock is posted exactly once with this record. Changes to recorded quantities require atomic reconciliation with stock and contribution summaries.'
}


Table donation_item {
  donation_item_id uuid [pk]
  donation_id uuid [not null]
  material_id uuid [not null]
  pledged_quantity decimal(12,2) [note: 'Optional promised quantity; NULL for direct unpledged receipt.']
  received_quantity decimal(12,2) [not null, default: 0, note: 'Total physically received, including unusable units.']
  unusable_quantity decimal(12,2) [not null, default: 0]
  item_note text

  indexes {
    (donation_id, material_id) [unique]
  }
  checks {
    `pledged_quantity IS NULL OR pledged_quantity > 0` [name: 'donation_item_positive_pledge']
    `received_quantity >= 0 AND unusable_quantity >= 0 AND unusable_quantity <= received_quantity` [name: 'donation_item_valid_receipt']
  }
  Note: 'Only for MATERIAL donation; enforce parent type in service or trigger. Quantities use material.material_unit, no independent donation unit. Usable received quantity = received_quantity - unusable_quantity. Saving the staff-entered receipt posts only usable quantity to material_transaction exactly once in the same transaction. Updates must reconcile the delta. pledged_quantity is legacy optional context, not a pledge workflow; leave NULL for direct receipts. Item receipt status is derived from quantities instead of independently stored.'
}

// 8. TASK EXECUTION & PROGRESS

Table task_progress {
  task_progress_id uuid [pk]
  task_id uuid [not null]
  submitted_by_user_id uuid [not null]
  progress_percent decimal(5,2) [not null]
  progress_note text
  submitted_at timestamp [not null]
  approved_by_user_id uuid [note: 'approvedBy: leader who manually approves completion. NULL until approval.']
  approved_at timestamp

  checks {
    `progress_percent >= 0 AND progress_percent <= 100` [name: 'task_progress_percent_range']
    `(approved_by_user_id IS NULL AND approved_at IS NULL) OR (approved_by_user_id IS NOT NULL AND approved_at IS NOT NULL AND progress_percent = 100 AND approved_at >= submitted_at)` [name: 'task_progress_completion_approval']
  }
  Note: '100 percent submission does not complete the task automatically. Service must authorize the responsible work-area leader and atomically set approval fields and task.task_status to COMPLETED. Prevent volunteers from setting approval fields.'
}

Table contribution_type {
  contribution_type_id uuid [pk]

  contribution_code varchar(50) [not null, unique]
  contribution_name varchar(100) [not null]
  contribution_description text
  contribution_status varchar(30)
}

Table contribution {
  contribution_id uuid [pk]
  contributed_by_user_id uuid [note: 'For non-donation contributions such as labor or purchase assistance. NULL for donations.']
  donation_id uuid [unique, note: 'One summary per donation. Donor identity and monetary amount come from donation.']
  season_id uuid [not null]
  contribution_type_id uuid [not null]
  contribution_description text
  contribution_date timestamp [not null]
  reference_type varchar(50)
  reference_id uuid

  checks {
    `(donation_id IS NOT NULL AND contributed_by_user_id IS NULL AND reference_type IS NULL AND reference_id IS NULL) OR (donation_id IS NULL AND contributed_by_user_id IS NOT NULL)` [name: 'contribution_single_source']
  }
  Note: 'Donation summaries are generated/upserted from staff-recorded actual receipts, never manually duplicate money or material quantities here. Validate matching season and contribution type (MONEY -> MONETARY_SUPPORT; MATERIAL -> MATERIAL_DONATION). For donation rows resolve identity via donation.donor_id and optional donor.user_id; for other contributions use contributed_by_user_id. Reports sum money from donation.amount only. Existing recognition_point is account-based: unlinked external donors appear in donor reports but do not receive account points.'
}
